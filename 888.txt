1-a
SELECT 
    d.dept_no,
    d.dept_name,
    COUNT(DISTINCT e.emp_no) AS employees_count
FROM departments d
INNER JOIN dept_emp e
    ON d.dept_no = e.dept_no
GROUP BY d.dept_no, d.dept_name;

1-b
SELECT d.dept_no, d.dept_name, 
count(DISTINCT e.emp_no) as employees_count
FROM departments d
INNER JOIN dept_emp e
ON d.dept_no = e.dept_no
GROUP BY d.dept_no, d.dept_name;

1-c
SELECT d.*, count(DISTINCT e.emp_no) FROM 
departments d 
LEFT OUTER JOIN dept_emp e 
ON d.dept_no = e.dept_no 
GROUP BY 1;

1-d
SELECT 
    d.dept_no,
    d.dept_name,
    COUNT(DISTINCT e.emp_no) AS employees_count
FROM dept_emp e
RIGHT OUTER JOIN departments d
    ON d.dept_no = e.dept_no
GROUP BY d.dept_no, d.dept_name;

2-a
SELECT
    d.dept_no,
    d.dept_name,
    e.emp_no,
    e.first_name,
    e.last_name,
    dm.from_date,
    dm.to_date
FROM departments d
LEFT JOIN dept_manager dm
    ON d.dept_no = dm.dept_no
    AND dm.to_date = '9999-01-01'
LEFT JOIN employees e
    ON dm.emp_no = e.emp_no
ORDER BY d.dept_no;




2-b
SELECT
    d.dept_no,
    d.dept_name,
    COUNT(DISTINCT de.emp_no) AS employees_count,
    COUNT(DISTINCT dm.emp_no) AS managers_count
FROM departments d
LEFT JOIN dept_emp de
    ON d.dept_no = de.dept_no
LEFT JOIN dept_manager dm
    ON d.dept_no = dm.dept_no
GROUP BY d.dept_no, d.dept_name
ORDER BY d.dept_no;



SELECT
    d.dept_no,
    d.dept_name,
    COUNT(DISTINCT de.emp_no) AS current_employees_count,
    COUNT(DISTINCT dm.emp_no) AS current_managers_count
FROM departments d
LEFT JOIN dept_emp de
    ON d.dept_no = de.dept_no
    AND de.to_date = '9999-01-01'
LEFT JOIN dept_manager dm
    ON d.dept_no = dm.dept_no
    AND dm.to_date = '9999-01-01'
GROUP BY d.dept_no, d.dept_name
ORDER BY d.dept_no;

2-c
SELECT
    d.dept_no,
    d.dept_name,
    COUNT(DISTINCT CASE 
        WHEN a.country = 'Poland' THEN de.emp_no 
    END) AS employees_living_in_poland
FROM departments d
LEFT JOIN dept_emp de
    ON d.dept_no = de.dept_no
LEFT JOIN adr_emp ae
    ON de.emp_no = ae.emp_no
LEFT JOIN address a
    ON ae.adr_no = a.adr_no
GROUP BY d.dept_no, d.dept_name
ORDER BY d.dept_no;



SELECT
    d.dept_no,
    d.dept_name,
    COUNT(DISTINCT CASE 
        WHEN a.country = 'Poland' THEN de.emp_no 
    END) AS current_employees_living_in_poland
FROM departments d
LEFT JOIN dept_emp de
    ON d.dept_no = de.dept_no
    AND de.to_date = '9999-01-01'
LEFT JOIN adr_emp ae
    ON de.emp_no = ae.emp_no
LEFT JOIN address a
    ON ae.adr_no = a.adr_no
GROUP BY d.dept_no, d.dept_name
ORDER BY d.dept_no;
