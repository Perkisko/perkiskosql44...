WHERE salary BETWEEN 52000 AND 53000;

ON salaries (salary);

/* =========================================================
   3a. Number of employees per department born in 1960
   ========================================================= */

SELECT
    d.dept_no,
    d.dept_name,
    COUNT(DISTINCT e.emp_no) AS number_of_employees
FROM departments d
JOIN dept_emp de
    ON d.dept_no = de.dept_no
JOIN employees e
    ON de.emp_no = e.emp_no
WHERE e.birth_date >= '1960-01-01'
  AND e.birth_date < '1961-01-01'
  AND de.to_date = '9999-01-01'
GROUP BY
    d.dept_no,
    d.dept_name
ORDER BY
    d.dept_no;

CREATE INDEX idx_employees_birth_empno
ON employees (birth_date, emp_no);

CREATE INDEX idx_dept_emp_emp_to_dept
ON dept_emp (emp_no, to_date, dept_no);

SELECT
    d.dept_no,
    d.dept_name,
    COUNT(DISTINCT e.emp_no) AS number_of_employees
FROM departments d
JOIN dept_emp de
    ON d.dept_no = de.dept_no
JOIN employees e
    ON de.emp_no = e.emp_no
WHERE e.birth_date >= '1960-01-01'
  AND e.birth_date < '1961-01-01'
  AND de.to_date = '9999-01-01'
GROUP BY
    d.dept_no,
    d.dept_name
ORDER BY
    d.dept_no;

/* =========================================================
   3c. Number of employees per department
       with last name starting from 'S'
   ========================================================= */

SELECT
    d.dept_no,
    d.dept_name,
    COUNT(DISTINCT e.emp_no) AS number_of_employees
FROM departments d
JOIN dept_emp de
    ON d.dept_no = de.dept_no
JOIN employees e
    ON de.emp_no = e.emp_no
WHERE e.last_name LIKE 'S%'
  AND de.to_date = '9999-01-01'
GROUP BY
    d.dept_no,
    d.dept_name
ORDER BY
    d.dept_no;

CREATE INDEX idx_employees_lastname_empno
ON employees (last_name, emp_no);

CREATE INDEX idx_dept_emp_emp_to_dept_2
ON dept_emp (emp_no, to_date, dept_no);

  AND de.to_date = '9999-01-01'
