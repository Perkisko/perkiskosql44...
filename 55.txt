/* ------------------------------------------------------------
   Task 1
   Assignment of employee to country and city
   Reusing the same view: current_emp_country
   ------------------------------------------------------------ */

CREATE OR REPLACE VIEW current_emp_country AS
SELECT
    e.emp_no,
    e.first_name,
    e.last_name,
    ae.adr_type,
    a.adr_no,
    a.country,
    a.city
FROM employees e
JOIN adr_emp ae
    ON e.emp_no = ae.emp_no
JOIN address a
    ON ae.adr_no = a.adr_no;


/* Reuse current_emp_country for countries */

SELECT
    emp_no,
    first_name,
    last_name,
    adr_type,
    country
FROM current_emp_country
ORDER BY emp_no;


/* Reuse current_emp_country for cities */

SELECT
    emp_no,
    first_name,
    last_name,
    adr_type,
    city
FROM current_emp_country
ORDER BY emp_no;


/* Reuse current_emp_country for both country and city */

SELECT
    emp_no,
    first_name,
    last_name,
    adr_type,
    country,
    city
FROM current_emp_country
ORDER BY emp_no;
