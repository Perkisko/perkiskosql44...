/* ------------------------------------------------------------
   Task 1
   Assignment of employee to address location: country and city
   ------------------------------------------------------------ */

CREATE OR REPLACE VIEW emp_location AS
SELECT
    e.emp_no,
    e.first_name,
    e.last_name,
    ae.adr_type,
    a.adr_no,
    a.country,
    a.city
FROM employees e
JOIN adr_emp ae
    ON e.emp_no = ae.emp_no
JOIN address a
    ON ae.adr_no = a.adr_no;


/* Reuse the same view for countries */

SELECT
    emp_no,
    first_name,
    last_name,
    adr_type,
    country
FROM emp_location
ORDER BY emp_no;


/* Reuse the same view for cities */

SELECT
    emp_no,
    first_name,
    last_name,
    adr_type,
    city
FROM emp_location
ORDER BY emp_no;


/* Reuse the same view for both country and city */

SELECT
    emp_no,
    first_name,
    last_name,
    adr_type,
    country,
    city
FROM emp_location
ORDER BY emp_no;
