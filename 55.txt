SELECT
    a.country,
    COUNT(DISTINCT e.emp_no) AS senior_engineers_count
FROM employees e
INNER JOIN titles t
    ON e.emp_no = t.emp_no
INNER JOIN adr_emp ae
    ON e.emp_no = ae.emp_no
INNER JOIN address a
    ON ae.adr_no = a.adr_no
WHERE t.title = 'Senior Engineer'
GROUP BY a.country
ORDER BY senior_engineers_count DESC;



SELECT
    d.dept_no,
    d.dept_name,

    COUNT(DISTINCT CASE
        WHEN t.title = 'Engineer' THEN de.emp_no
    END) AS engineers_count,

    COUNT(DISTINCT dm.emp_no) AS managers_count

FROM departments d

LEFT JOIN dept_emp de
    ON d.dept_no = de.dept_no
    AND de.to_date = '9999-01-01'

LEFT JOIN titles t
    ON de.emp_no = t.emp_no
    AND t.to_date = '9999-01-01'

LEFT JOIN dept_manager dm
    ON d.dept_no = dm.dept_no
    AND dm.to_date = '9999-01-01'

GROUP BY d.dept_no, d.dept_name
ORDER BY d.dept_no;
