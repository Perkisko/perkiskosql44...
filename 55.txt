SELECT 'address' AS table_name, COUNT(*) AS columns_count
FROM INFORMATION_SCHEMA.COLUMNS
WHERE TABLE_SCHEMA = 'employees'
  AND TABLE_NAME = 'address'

UNION ALL

SELECT 'adr_emp', COUNT(*)
FROM INFORMATION_SCHEMA.COLUMNS
WHERE TABLE_SCHEMA = 'employees'
  AND TABLE_NAME = 'adr_emp'

UNION ALL

SELECT 'departments', COUNT(*)
FROM INFORMATION_SCHEMA.COLUMNS
WHERE TABLE_SCHEMA = 'employees'
  AND TABLE_NAME = 'departments'

UNION ALL

SELECT 'dept_emp', COUNT(*)
FROM INFORMATION_SCHEMA.COLUMNS
WHERE TABLE_SCHEMA = 'employees'
  AND TABLE_NAME = 'dept_emp'

UNION ALL

SELECT 'dept_manager', COUNT(*)
FROM INFORMATION_SCHEMA.COLUMNS
WHERE TABLE_SCHEMA = 'employees'
  AND TABLE_NAME = 'dept_manager'

UNION ALL

SELECT 'employees', COUNT(*)
FROM INFORMATION_SCHEMA.COLUMNS
WHERE TABLE_SCHEMA = 'employees'
  AND TABLE_NAME = 'employees'

UNION ALL

SELECT 'salaries', COUNT(*)
FROM INFORMATION_SCHEMA.COLUMNS
WHERE TABLE_SCHEMA = 'employees'
  AND TABLE_NAME = 'salaries'

UNION ALL

SELECT 'titles', COUNT(*)
FROM INFORMATION_SCHEMA.COLUMNS
WHERE TABLE_SCHEMA = 'employees'
  AND TABLE_NAME = 'titles';





SELECT TABLE_NAME AS table_name, COUNT(*) AS columns_count
FROM INFORMATION_SCHEMA.COLUMNS
WHERE TABLE_SCHEMA = 'employees'
  AND TABLE_NAME IN (
      'address',
      'adr_emp',
      'departments',
      'dept_emp',
      'dept_manager',
      'employees',
      'salaries',
      'titles'
  )
GROUP BY TABLE_NAME
ORDER BY TABLE_NAME;


SELECT TABLE_NAME, COUNT(*) AS columns_count

WHERE TABLE_SCHEMA = 'employees'
