SELECT 'Total' AS city, COUNT(DISTINCT ae.emp_no) AS employees_count
FROM adr_emp ae
INNER JOIN dept_emp de
    ON ae.emp_no = de.emp_no
WHERE de.to_date = '9999-01-01'

UNION

SELECT a.city AS city, COUNT(DISTINCT ae.emp_no) AS employees_count
FROM address a
INNER JOIN adr_emp ae
    ON a.adr_no = ae.adr_no
INNER JOIN dept_emp de
    ON ae.emp_no = de.emp_no
WHERE de.to_date = '9999-01-01'
GROUP BY a.city;








SELECT '< 30' AS age_range, COUNT(DISTINCT e.emp_no) AS employees_count
FROM employees e
INNER JOIN dept_emp de
    ON e.emp_no = de.emp_no
WHERE de.to_date = '9999-01-01'
  AND TIMESTAMPDIFF(YEAR, e.birth_date, CURDATE()) < 30

UNION

SELECT '30 - 39' AS age_range, COUNT(DISTINCT e.emp_no) AS employees_count
FROM employees e

    ON e.emp_no = de.emp_no
WHERE de.to_date = '9999-01-01'
  AND TIMESTAMPDIFF(YEAR, e.birth_date, CURDATE()) BETWEEN 30 AND 39

UNION
