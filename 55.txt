WITH RECURSIVE dept_tree AS
(
    SELECT
        dorg.dept_no AS root_dept_no,
        dorg.dept_no AS dept_no,
        1 AS hierarchy_level
    FROM dept_org dorg
    WHERE dorg.org_no IS NULL

    UNION ALL

    SELECT
        parent.root_dept_no,
        child.dept_no,
        parent.hierarchy_level + 1 AS hierarchy_level
    FROM dept_tree parent
    INNER JOIN dept_org child
        ON child.org_no = parent.dept_no
)
SELECT
    dt.root_dept_no,
    root_dept.dept_name AS root_dept_name,
    dt.dept_no AS child_dept_no,
    child_dept.dept_name AS child_dept_name,
    dt.hierarchy_level,
    COUNT(DISTINCT de.emp_no) AS number_of_employees
FROM dept_tree dt
INNER JOIN departments root_dept
    ON dt.root_dept_no = root_dept.dept_no
INNER JOIN departments child_dept
    ON dt.dept_no = child_dept.dept_no
LEFT JOIN dept_emp de
    ON dt.dept_no = de.dept_no
   AND de.to_date = '9999-01-01'
GROUP BY
    dt.root_dept_no,
    root_dept.dept_name,
    dt.dept_no,
    child_dept.dept_name,
    dt.hierarchy_level
ORDER BY
    dt.root_dept_no,
    dt.hierarchy_level,
    dt.dept_no;








WITH RECURSIVE dept_tree AS
(
    SELECT
        dorg.dept_no AS root_dept_no,
        dorg.dept_no AS dept_no
    FROM dept_org dorg
    WHERE dorg.org_no IS NULL

    UNION ALL

    SELECT
        parent.root_dept_no,
        child.dept_no
    FROM dept_tree parent
    INNER JOIN dept_org child
        ON child.org_no = parent.dept_no
)
SELECT
    dt.root_dept_no,
    root_dept.dept_name AS root_dept_name,
    GROUP_CONCAT(
        DISTINCT child_dept.dept_name
        ORDER BY child_dept.dept_name
        SEPARATOR ', '
    ) AS departments_under_root,
    COUNT(DISTINCT de.emp_no) AS number_of_employees
FROM dept_tree dt
INNER JOIN departments root_dept
    ON dt.root_dept_no = root_dept.dept_no
INNER JOIN departments child_dept
    ON dt.dept_no = child_dept.dept_no
LEFT JOIN dept_emp de
    ON dt.dept_no = de.dept_no
   AND de.to_date = '9999-01-01'
GROUP BY
    dt.root_dept_no,
    root_dept.dept_name
ORDER BY
    dt.root_dept_no;







WITH RECURSIVE dept_tree AS
(
    SELECT
        do.dept_no AS root_dept_no,
        d.dept_name AS root_dept_name,
        do.dept_no,
        d.dept_name,
        do.org_no,
        1 AS depth_level,
        CAST(do.dept_no AS CHAR(500)) AS dept_path,
        CAST(d.dept_name AS CHAR(1000)) AS dept_name_path
    FROM dept_org do
    JOIN departments d
        ON do.dept_no = d.dept_no
    WHERE do.org_no IS NULL

    UNION ALL

    SELECT
        parent.root_dept_no,
        parent.root_dept_name,
        child.dept_no,
        d.dept_name,
        child.org_no,
        parent.depth_level + 1 AS depth_level,
        CONCAT(parent.dept_path, ' -> ', child.dept_no) AS dept_path,
        CONCAT(parent.dept_name_path, ' -> ', d.dept_name) AS dept_name_path
    FROM dept_tree parent
    JOIN dept_org child
        ON child.org_no = parent.dept_no
    JOIN departments d
        ON child.dept_no = d.dept_no
)
SELECT
    dt.root_dept_no,
    dt.root_dept_name,
    dt.dept_no,
    dt.dept_name,
    dt.depth_level,
    dt.dept_path,
    dt.dept_name_path,
    COUNT(DISTINCT de.emp_no) AS total_employees
FROM dept_tree dt
LEFT JOIN dept_emp de
    ON de.dept_no = dt.dept_no
    AND de.to_date = '9999-01-01'
GROUP BY
    dt.root_dept_no,
    dt.root_dept_name,
    dt.dept_no,
    dt.dept_name,
    dt.depth_level,
    dt.dept_path,
    dt.dept_name_path
ORDER BY
    dt.dept_path;

WITH RECURSIVE dept_name_hierarchy AS
(
    SELECT
        dorg.dept_no,
        d.dept_name,
        dorg.org_no AS parent_dept_no,
        1 AS hierarchy_level,
        CAST(d.dept_name AS CHAR(1000)) AS department_hierarchy
    FROM dept_org dorg
    INNER JOIN departments d
        ON dorg.dept_no = d.dept_no
    WHERE dorg.org_no IS NULL

    UNION ALL

    SELECT
        child.dept_no,
        d.dept_name,
        child.org_no AS parent_dept_no,
        parent.hierarchy_level + 1 AS hierarchy_level,
        CONCAT(
            parent.department_hierarchy,
            ' -> ',
            d.dept_name
        ) AS department_hierarchy
    FROM dept_name_hierarchy parent
    INNER JOIN dept_org child
        ON child.org_no = parent.dept_no
    INNER JOIN departments d
        ON child.dept_no = d.dept_no
)
SELECT
    dept_no,
    dept_name,
    parent_dept_no,
    hierarchy_level,
    department_hierarchy
FROM dept_name_hierarchy
ORDER BY department_hierarchy;
