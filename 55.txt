/* ------------------------------------------------------------
   Task 2a
   Selecting the oldest and youngest employee in each department
   Reusing the same view for both oldest and youngest
   ------------------------------------------------------------ */


/* View containing employees, their current departments, and age */

CREATE OR REPLACE VIEW emp_dept_age AS
SELECT
    e.emp_no,
    e.first_name,
    e.last_name,
    e.birth_date,
    TIMESTAMPDIFF(YEAR, e.birth_date, CURDATE()) AS age,
    d.dept_no,
    d.dept_name
FROM employees e
JOIN current_dept_emp cde
    ON e.emp_no = cde.emp_no
JOIN departments d
    ON cde.dept_no = d.dept_no;


/* Reuse the same view to get the oldest employee in each department */

SELECT
    eda.dept_no,
    eda.dept_name,
    eda.emp_no,
    eda.first_name,
    eda.last_name,
    eda.birth_date,
    eda.age,
    'OLDEST' AS age_type
FROM emp_dept_age eda
WHERE eda.birth_date = (
    SELECT MIN(eda2.birth_date)
    FROM emp_dept_age eda2
    WHERE eda2.dept_no = eda.dept_no
)

UNION

/* Reuse the same view to get the youngest employee in each department */

SELECT
    eda.dept_no,
    eda.dept_name,
    eda.emp_no,
    eda.first_name,
    eda.last_name,
    eda.birth_date,
    eda.age,
    'YOUNGEST' AS age_type
FROM emp_dept_age eda
WHERE eda.birth_date = (
    SELECT MAX(eda2.birth_date)
    FROM emp_dept_age eda2
    WHERE eda2.dept_no = eda.dept_no
)




SET SESSION net_read_timeout = 120;
SET SESSION net_write_timeout = 120;
SET SESSION wait_timeout = 600;
SET SESSION interactive_timeout = 600;

ORDER BY dept_no, age_type, birth_date;
