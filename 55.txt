WITH RECURSIVE numbers AS
(
    SELECT 7 AS n

    UNION ALL

    SELECT n + 1
    FROM numbers
    WHERE n < 37
)
SELECT *
FROM numbers;














WITH hired_before_year AS
(
    SELECT
        emp_no,
        first_name,
        last_name,
        gender,
        birth_date,
        hire_date
    FROM employees
    WHERE YEAR(hire_date) < 1990
)
SELECT
    emp_no,
    first_name,
    last_name,
    gender,
    birth_date,
    hire_date
FROM hired_before_year
ORDER BY hire_date;







/* ------------------------------------------------------------
   Task 2c
   Finding employees who worked in departments d004 and d005
   and are living in the same city: Istanbul
   Reusing the same view
   ------------------------------------------------------------ */


/* View containing employees, department history, and addresses */

CREATE OR REPLACE VIEW emp_dept_address AS
SELECT
    e.emp_no,
    e.first_name,
    e.last_name,
    de.dept_no,
    de.from_date AS dept_from_date,
    de.to_date AS dept_to_date,
    ae.adr_type,
    a.adr_no,
    a.city,
    a.street,
    a.house_no,
    a.post_code,
    a.country
FROM employees e
JOIN dept_emp de
    ON e.emp_no = de.emp_no
JOIN adr_emp ae
    ON e.emp_no = ae.emp_no
JOIN address a
    ON ae.adr_no = a.adr_no;


/* Reuse the same view:
   employees who worked in both d004 and d005
   and live in Istanbul */

SELECT
    eda.emp_no,
    eda.first_name,
    eda.last_name,
    eda.city,
    eda.country,
    eda.street,
    eda.house_no,
    eda.post_code
FROM emp_dept_address eda
WHERE eda.dept_no IN ('d004', 'd005')
  AND eda.city = 'Istanbul'
GROUP BY
    eda.emp_no,
    eda.first_name,
    eda.last_name,
    eda.city,
    eda.country,
    eda.street,
    eda.house_no,
    eda.post_code
HAVING COUNT(DISTINCT eda.dept_no) = 2
ORDER BY eda.emp_no;
