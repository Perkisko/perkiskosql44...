    WITH RECURSIVE dept_tree AS
(
    SELECT
        dorg.dept_no AS root_dept_no,
        dorg.dept_no AS dept_no
    FROM dept_org dorg
    WHERE dorg.org_no IS NULL

    UNION ALL

    SELECT
        parent.root_dept_no,
        child.dept_no
    FROM dept_tree parent
    INNER JOIN dept_org child
        ON child.org_no = parent.dept_no
)
SELECT
    dt.root_dept_no,
    d.dept_name AS root_dept_name,
    COUNT(DISTINCT de.emp_no) AS number_of_employees
FROM dept_tree dt
INNER JOIN departments d
    ON dt.root_dept_no = d.dept_no
LEFT JOIN dept_emp de
    ON dt.dept_no = de.dept_no
   AND de.to_date = '9999-01-01'
GROUP BY
    dt.root_dept_no,
    d.dept_name
ORDER BY
    dt.root_dept_no;




WITH RECURSIVE dept_name_hierarchy AS
(
    SELECT
        dorg.dept_no,
        d.dept_name,
        dorg.org_no AS parent_dept_no,
        1 AS hierarchy_level,
        CAST(d.dept_name AS CHAR(1000)) AS department_hierarchy
    FROM dept_org dorg
    INNER JOIN departments d
        ON dorg.dept_no = d.dept_no
    WHERE dorg.org_no IS NULL

    UNION ALL

    SELECT
        child.dept_no,
        d.dept_name,
        child.org_no AS parent_dept_no,
        parent.hierarchy_level + 1 AS hierarchy_level,
        CONCAT(
            parent.department_hierarchy,
            ' -> ',
            d.dept_name
        ) AS department_hierarchy
    FROM dept_name_hierarchy parent
    INNER JOIN dept_org child
        ON child.org_no = parent.dept_no
    INNER JOIN departments d
        ON child.dept_no = d.dept_no
)
SELECT
    dept_no,
    dept_name,
    parent_dept_no,
    hierarchy_level,
    department_hierarchy
FROM dept_name_hierarchy
ORDER BY department_hierarchy;
